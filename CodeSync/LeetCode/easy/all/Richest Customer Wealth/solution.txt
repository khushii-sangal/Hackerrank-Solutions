class Solution {
public:
    int maximumWealth(vector<vector<int>>& accounts) {
        int sum=0,wealth=0;
        for(int i=0;i<accounts.size();i++)
        {
            for(int j=0;i<accounts[i].size();j++)
            {
                sum+=accounts[j];
            }
            wealth=max(sum,wealth);
        }
        return wealth;
    }
};
